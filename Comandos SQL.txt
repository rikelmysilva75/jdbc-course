create table employees (              //---> cria uma tabela com nome employees
	employee_id int,              //---> cria uma coluna com nome employee_id que pode receber números
	first_name varchar(50),       //---> cria uma coluna com nome first_name que pode ter até 50 caracteres
	last_name varchar(50),	      //---> cria uma coluna com nome last_name que pode ter até 50 caracteres
	hourly_pay decimal(5,2),      //---> cria uma coluna com nome hourly_pay que pode ter números com 5 casas decimais e 2 digitos depois da virgula
	hire_date date,               //---> cria uma coluna com nome hire_date que pode receber datas
);
----------------------------------------------------------------------------------------------------------------------------------------------------------------------------
